	$(document).ready(function () {
		$('.nav-tabs > li a[title]').tooltip();
		$('a[data-toggle="tab"]').on('shown.bs.tab', function (e) {
			var target = $(e.target);
			if (target.parent().hasClass('disabled')) {
				return false;
			}
		});
		$(".next-step").click(function (e) {
			var control = 0;
			var id = $(this).data("id");
			if (id==7) {
				$('#stepBox').css('display','none');
			}else{
				$('#step' +id+ ' .form-group').each(function(i, obj) {
					if($(obj).find('input').is(':checked')){
						control = 1;
					}
				});
				if (control==1) {
					if (id==6) {
						if($('#check_6_yes').is(':checked')){
							if ($('#fullname').val()=="" || $('#phone').val()=="") {
								$('.pollError').css('display','block');
								$('.pollError').html('Lütfen boş alan bırakmayınız!');
								setTimeout(function(){ 

									$('#stepBox').scrollTop(0);

								}, 100);
							}else{
								if ($('#check_aggrement_1').is(':checked') && $('#check_aggrement_2').is(':checked')) {
									$('.pollError').css('display','none');
									var active = $('.wizard .nav-tabs li.active');
									active.next().removeClass('disabled');
									nextTab(active);
									$('.wizard-inner').css('display','none');
									$.ajax({
										url: "contact/poll",
										type: "post",
										data: $('.poll-form').serialize(),
										success: function (response) {
										},
										error: function(jqXHR, textStatus, errorThrown) {
											console.log(textStatus, errorThrown);
										}
									});
								}else{
									$('.pollError').css('display','block');
									$('.pollError').html('Sözleşmeleri kabul etmeden devam edemezsiniz!');
									setTimeout(function(){ 

										$('#stepBox').scrollTop(0);

									}, 100);
								}
							}
						}else{
							$('.pollError').css('display','none');
							var active = $('.wizard .nav-tabs li.active');
							active.next().removeClass('disabled');
							nextTab(active);
							$('.wizard-inner').css('display','none');
							$.ajax({
								url: "contact/poll",
								type: "post",
								data: $('.poll-form').serialize(),
								success: function (response) {
								},
								error: function(jqXHR, textStatus, errorThrown) {
									console.log(textStatus, errorThrown);
								}
							});
						}
					}else{
						$('.pollError').css('display','none');
						var active = $('.wizard .nav-tabs li.active');
						active.next().removeClass('disabled');
						nextTab(active);
					}
				}else{
					$('.pollError').css('display','block');
					$('.pollError').html('Seçim yapmadan devam edemezsiniz!');
					setTimeout(function(){ 

						$('#stepBox').scrollTop(0);

					}, 100);
				}
			}
		});
		$(".prev-step").click(function (e) {
			$('.pollError').css('display','none');
			var active = $('.wizard .nav-tabs li.active');
			prevTab(active);
		});
	});
	function nextTab(elem) {
		$(elem).next().find('a[data-toggle="tab"]').click();
	}
	function prevTab(elem) {
		$(elem).prev().find('a[data-toggle="tab"]').click();
	}
	$('.nav-tabs').on('click', 'li', function() {
